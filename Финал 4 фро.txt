    //11. Хранилище данных меню в JavaScript
    let cachedMenu = [];

    //12. Функция загрузки и вывода меню
    async function loadMenu() {
        const search = document.getElementById('search-input').value;
        const res = await fetch(`/api/menu?search=${encodeURIComponent(search)}`);
        cachedMenu = await res.json();

        const list = document.getElementById('menu-list');
        list.innerHTML = '';

        if (cachedMenu.length === 0) {
            list.innerHTML = '<p style="color: var(--text-muted); grid-column: 1/-1;">Ничего не найдено</p>';
            return;
        }

        const sections = [
            { name: "Блюда", types: ["пицца", "горячее", "суп", "салат"], emoji: "🍽️" },
            { name: "Закуски", types: ["закуски"], emoji: "🍟" },
            { name: "Напитки", types: ["напитки"], emoji: "🥤" }
        ];

        sections.forEach(cat => {
            const items = cachedMenu.filter(i => cat.types.includes(i.type));

            if (items.length > 0) {
                list.innerHTML += '<h3 style="grid-column: 1/-1; font-size: 26px; margin-top: 10px; margin-bottom: 10px; font-weight: 700; color: var(--text-main); text-align: left; width: 100%;">' + cat.name + '</h3>';

                items.forEach(item => {
                    list.innerHTML +=
                        '<div class="product-card">' +
                            '<div class="product-img-wrapper">' +
                                '<img src="' + item.img + '" alt="' + item.name + '" class="product-pic" onclick="showDetails(' + item.id + ')" onerror="this.style.display=\'none\'; this.nextElementSibling.style.display=\'block\';">' +
                                '<div class="product-no-pic" style="display:none; font-size:44px; text-align:center; padding-top:40px;">' + cat.emoji + '</div>' +
                            '</div>' +
                            '<div class="product-info">' +
                                '<h4 onclick="showDetails(' + item.id + ')">' + item.name + '</h4>' +
                                '<div class="product-desc">' + item.desc + '</div>' +
                            '</div>' +
                            '<div class="product-footer">' +
                                '<div class="price">' + item.price + ' ₽</div>' +
                                '<div style="display: flex; align-items: center;">' +
                                    '<div class="dodo-qty-box">' +
                                        '<button class="dodo-qty-btn" onclick="const i=document.getElementById(\'qty-' + item.id + '\'); if(i.value>1)i.value--;">−</button>' +
                                        '<input type="number" id="qty-' + item.id + '" value="1" min="1" class="qty-input" style="margin-right:0; border:none; width:28px; background:transparent; text-align:center; color:white; font-size:14px; font-weight:600; outline:none;" readonly>' +
                                        '<button class="dodo-qty-btn" onclick="const i=document.getElementById(\'qty-' + item.id + '\'); i.value++;">+</button>' +
                                    '</div>' +
                                    '<button class="btn" style="padding: 10px 16px; font-size:14px; border-radius:10px;" onclick="addToCart(' + item.id + ')">В корзину</button>' +
                                '</div>' +
                            '</div>' +
                        '</div>';
                });
            }
        });
    }

    //13. Функция переключения веса и расчета БЖУ
    function toggleDodoWeights(mode, id) {
        const item = cachedMenu.find(p => p.id === id);
        if(!item) return;

        const btn100 = document.getElementById('tab-100-' + id);
        const btnAll = document.getElementById('tab-all-' + id);

        const ccalEl = document.getElementById('dodo-ccal-' + id);
        const bEl = document.getElementById('dodo-b-' + id);
        const jEl = document.getElementById('dodo-j-' + id);
        const uEl = document.getElementById('dodo-u-' + id);

        if(mode === '100') {
            btn100.classList.add('active');
            btnAll.classList.remove('active');
            ccalEl.innerText = item.ccal;
            bEl.innerText = item.b;
            jEl.innerText = item.j;
            uEl.innerText = item.u;
        } else {
            btn100.classList.remove('active');
            btnAll.classList.add('active');
            const multiplier = item.weight / 100;
            ccalEl.innerText = Math.round(item.ccal * multiplier);
            bEl.innerText = (item.b * multiplier).toFixed(1);
            jEl.innerText = (item.j * multiplier).toFixed(1);
            uEl.innerText = (item.u * multiplier).toFixed(1);
        }
    }

    //14. Функция показа подробной информации о блюде

            '<div class="details-modal">' +

                '<p class="dodo-main-desc">' + item.desc + '</p>' +
                '<div class="dodo-box">' +
                    '<div class="dodo-toggle">' +

                        '<button id="tab-all-' + item.id + '" class="dodo-tab" onclick="toggleDodoWeights(\'all\', ' + item.id + ')">Вся ' + labelUnit + '</button>' +

    //15. Функция добавления товара в корзину
    async function addToCart(id) {
        const quantity = parseInt(document.getElementById('qty-' + id).value) || 1;
        await fetch('/api/cart', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ id: id, quantity: quantity })
        });
        document.getElementById('qty-' + id).value = "1";
        loadCart();
    }

    //16. Функция отображения корзины и расчета общих БЖУ
    async function loadCart() {
        const res = await fetch('/api/cart');
        const data = await res.json();
        const list = document.getElementById('cart-list');

        document.getElementById('cart-total').innerText = data.total;
        document.getElementById('cart-count').innerText = data.items.reduce((s, i) => s + i.quantity, 0);

        if(data.items.length === 0) {
            list.innerHTML = "Корзина пуста";
            return;
        }

        let totalCcal = 0, totalB = 0, totalJ = 0, totalU = 0;
        list.innerHTML = '';

        data.items.forEach(item => {
            const multiplier = (item.weight || 100) / 100;
            totalCcal += Math.round((item.ccal || 0) * multiplier * item.quantity);
            totalB += (item.b || 0) * multiplier * item.quantity;
            totalJ += (item.j || 0) * multiplier * item.quantity;
            totalU += (item.u || 0) * multiplier * item.quantity;

            const itemEl = document.createElement('div');
            itemEl.className = 'cart-item';
            itemEl.innerHTML =
                '<div>' +
                    '<div style="font-weight:600;">' + item.name + '</div>' +
                    '<div style="font-size:12px; color:var(--text-muted);">' + item.price + ' ₽ × ' + item.quantity + '</div>' +
                '</div>' +
                '<div class="cart-item-controls">' +
                    '<button onclick="updateQty(' + item.id + ', \'minus\')">-</button>' +
                    '<span style="margin: 0 4px; font-weight:600;">' + item.quantity + '</span>' +
                    '<button onclick="updateQty(' + item.id + ', \'plus\')">+</button>' +
                    '<button style="background:none; border:none; color:var(--text-muted); font-size:20px; cursor:pointer; line-height:1; transition:color 0.2s;" onmouseover="this.style.color=\'var(--primary-color)\'" onmouseout="this.style.color=\'var(--text-muted)\'" onclick="updateQty(' + item.id + ', \'delete\')">×</button>' +
                '</div>';
            list.appendChild(itemEl);
        });

        const summaryEl = document.createElement('div');
        summaryEl.style.cssText = "margin-top: 15px; padding-top: 10px; border-top: 1px dashed var(--border-color); font-size: 11px; color: var(--text-muted);";
        summaryEl.innerHTML =
            '<div style="font-weight:600; margin-bottom:5px; color:var(--text-main);">Итого БЖУ заказа:</div>' +
            '<div style="display:grid; grid-template-columns: repeat(4, 1fr); text-align:center; background:#333333; padding:6px; border-radius:8px; border:1px solid var(--border-color);">' +
                '<div>🔥 ' + Math.round(totalCcal) + '<br>ккал</div>' +
                '<div>' + totalB.toFixed(1) + 'г<br>Б</div>' +
                '<div>' + totalJ.toFixed(1) + 'г<br>Ж</div>' +
                '<div>' + totalU.toFixed(1) + 'г<br>У</div>' +
            '</div>';
        list.appendChild(summaryEl);
    }

    //17. Функция изменения количества товара в корзине
    async function updateQty(id, action) {
        await fetch('/api/cart', {
            method: 'PUT',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ id: id, action: action })
        });
        loadCart();
    }

    //18. Функция сброса поиска
    function resetFilters() { document.getElementById('search-input').value = ''; loadMenu(); }

    //19. Функция загрузки истории заказов
    async function loadHistory() {
        const res = await fetch('/api/history');
        const history = await res.json();
        const list = document.getElementById('history-list');
        list.innerHTML = history.length === 0 ? '<p style="color: var(--text-muted); font-size:13px; margin:0;">Нет заказов</p>' : '';

        history.forEach(h => {
            const histEl = document.createElement('div');
            histEl.style.cssText = "display:flex; justify-content:space-between; align-items:center; margin-bottom:8px; font-size:14px;";
            histEl.innerHTML = '<span>Заказ <b>' + h.id + '</b></span><button class="btn btn-secondary" style="padding:4px 10px; font-size:12px; border-radius:6px;" onclick="repeatOrder(\'' + h.id + '\')">Повторить</button>';
            list.appendChild(histEl);
        });
    }

    //20. Функция повторения старого заказа
    async function repeatOrder(orderId) {
        const res = await fetch('/api/history/repeat', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ orderId: orderId })
        });
        if (res.ok) {
            loadCart();
        }
    }

    //21. Функция отправки и проверки заказа
    async function checkout() {
        const errBox = document.getElementById('order-error');
        errBox.style.display = 'none';
        errBox.innerText = '';
        document.getElementById('order-success').innerHTML = '';

        const address = document.getElementById('order-address').value;
        const promo = document.getElementById('order-promo').value;

        const res = await fetch('/api/order', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ address: address, promo: promo })
        });
        const data = await res.json();

        if(!res.ok) {
            errBox.innerText = data.error;
            errBox.style.display = 'block';
        } else {
            document.getElementById('order-success').innerHTML = '<div class="success-box">🎉 Заказ №' + data.orderId + ' оформлен! С учетом скидки к оплате: <b>' + data.total + ' ₽</b>. Доставка через 40 минут.</div>';
            document.getElementById('order-address').value = '';
            document.getElementById('order-promo').value = '';
            loadCart();
            loadHistory();
        }
    }

    //22. Первоначальный запуск приложения
    loadMenu();
    loadCart();
    loadHistory();
